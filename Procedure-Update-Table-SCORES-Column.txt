CREATE PROCEDURE ChangeScores
(
    IN NumScore1 INTEGER,
    IN NumScore2 INTEGER,
    IN SummaScoreIN INTEGER
)
LANGUAGE plpgsql AS $$
DECLARE
    SummaScoreF INTEGER = 0;
BEGIN

    SELECT SCORES.SummaScore INTO SummaScoreF FROM SCORES WHERE SCORES.NumScore = NumScore1;
   -- BEGIN TRANSACTION;
    IF SummaScoreF>0 THEN
       UPDATE SCORES SET SummaScore = SummaScore + SummaScoreIN
       WHERE NumScore = NumScore2;    
       COMMIT;       
    END IF;  
END; $$;

CALL ChangeScores(1,2,300);
