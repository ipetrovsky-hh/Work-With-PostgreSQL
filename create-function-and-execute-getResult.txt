-- create
CREATE TABLE EMPLOYEE (
  empId INTEGER PRIMARY KEY,
  name TEXT NOT NULL,
  dept TEXT NOT NULL
);

-- insert
INSERT INTO EMPLOYEE VALUES (1001, 'Clark', 'Sales');
INSERT INTO EMPLOYEE VALUES (1002, 'Dave', 'Accounting');
INSERT INTO EMPLOYEE VALUES (1003, 'Ava', 'Sales');

-- fetch 
SELECT * FROM EMPLOYEE WHERE dept = 'Sales';

CREATE OR REPLACE FUNCTION get_EMPLOYEE_total(p_dept TEXT)
RETURNS NUMERIC AS $$
    SELECT COALESCE(SUM(1), 0)
    FROM EMPLOYEE
    WHERE dept = p_dept;
$$ LANGUAGE sql STABLE;

-- Вызов
SELECT get_EMPLOYEE_total('Sales');
